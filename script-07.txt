/* ---------- play ---------- */
let timerId = null, remaining = 0, running = false;
function fmt(s){ return Math.floor(s/60) + ":" + String(s%60).padStart(2,'0'); }
function paintClock(){
  $('clock').textContent = fmt(remaining);
  $('clock').classList.toggle('warn', remaining <= 30 && remaining > 0);
}
function stopTimer(){ clearInterval(timerId); timerId = null; running = false; $('timerStart').textContent = "▶ START"; }
function toggleTimer(){
  if(running){ stopTimer(); return; }
  running = true; $('timerStart').textContent = "⏸ PAUSE";
  timerId = setInterval(() => {
    remaining--; paintClock();
    if(remaining <= 0){
      stopTimer(); $('clock').textContent = "TIME!";
      beep(880,200); setTimeout(()=>beep(660,300),240);
    }
  }, 1000);
}
$('timerStart').onclick = toggleTimer;
$('timerReset').onclick = () => { stopTimer(); remaining = state.timerSeconds; paintClock(); };

function showPlay(){
  const rot = (state.roundNo - 1) % state.playerCount;
  const order = state.order.slice(rot).concat(state.order.slice(0, rot));
  $('roundPill').textContent = "CLUE ROUND " + state.roundNo + " OF " + state.clueRounds;
  const list = $('orderList');
  list.innerHTML = "";
  order.forEach((p, idx) => {
    const li = document.createElement('li');
    if(idx === 0) li.classList.add('first');
    li.innerHTML = `<span class="num">${idx+1}</span><span>${playerName(p)}</span>` +
      (idx === 0 ? '<span class="starts">STARTS 👈</span>' : '');
    list.appendChild(li);
  });
  stopTimer();
  if(state.timerSeconds === 0) $('timerPanel').classList.add('hidden');
  else { $('timerPanel').classList.remove('hidden'); remaining = state.timerSeconds; paintClock(); }
  $('nextPhaseBtn').textContent = state.roundNo < state.clueRounds ? "➡️ NEXT CLUE ROUND" : "🗳️ TIME TO VOTE";
  show('s-play');
}
$('nextPhaseBtn').onclick = () => {
  stopTimer();
  if(state.roundNo < state.clueRounds){ state.roundNo++; showPlay(); }
  else if(state.voteMode === "secret") startBallots();
  else showQuickVote();
};

/* ---------- quick vote ---------- */
function showQuickVote(){
  const grid = $('voteGrid');
  grid.innerHTML = "";
  state.picks = [];
  $('voteHint').textContent = state.imposterCount === 1
    ? "Count down from three and everyone point at once. Then tap that player."
    : "There are two imposters. Tap the two players who got pointed at most.";
  for(let i=0;i<state.playerCount;i++){
    const b = document.createElement('button');
    b.className = "vote-btn"; b.textContent = playerName(i);
    b.onclick = () => {
      const at = state.picks.indexOf(i);
      if(at > -1) state.picks.splice(at,1);
      else if(state.picks.length < state.imposterCount) state.picks.push(i);
      grid.querySelectorAll('.vote-btn').forEach((btn,idx) => btn.classList.toggle('picked', state.picks.includes(idx)));
      $('revealBtn').disabled = state.picks.length !== state.imposterCount;
    };
    grid.appendChild(b);
  }
  $('revealBtn').disabled = true;
  show('s-vote');
}
$('revealBtn').onclick = () => { state.accused = state.picks.slice(); showResult(); };

/* ---------- secret ballot ---------- */
function startBallots(){
  state.ballots = new Array(state.playerCount).fill(null);
  state.voterIndex = 0;
  showBallot();
}
function showBallot(){
  const v = state.voterIndex;
  $('ballotName').textContent = playerName(v);
  $('ballotAsk').classList.remove('hidden');
  $('ballotGrid').classList.remove('hidden');
  $('ballotLock').classList.add('hidden');
  $('ballotNext').classList.add('hidden');
  const grid = $('ballotGrid');
  grid.innerHTML = "";
  for(let i=0;i<state.playerCount;i++){
    const b = document.createElement('button');
    b.className = "vote-btn"; b.textContent = playerName(i);
    if(i === v){ b.disabled = true; b.textContent = playerName(i) + " (you)"; }
    b.onclick = () => {
      state.ballots[v] = i;
      $('ballotAsk').classList.add('hidden');
      $('ballotGrid').classList.add('hidden');
      $('ballotLock').classList.remove('hidden');
      $('ballotNext').classList.remove('hidden');
      beep(540,90);
    };
    grid.appendChild(b);
  }
  show('s-ballot');
}
$('ballotNext').onclick = () => {
  state.voterIndex++;
  if(state.voterIndex >= state.playerCount) countBallots();
  else showBallot();
};
function voteCounts(){
  const counts = new Array(state.playerCount).fill(0);
  state.ballots.forEach(v => { if(v != null) counts[v]++; });
  return counts;
}
function renderTally(el){
  const counts = voteCounts();
  const max = Math.max(1, ...counts);
  el.innerHTML = counts.map((c,i) => `
    <div class="tallyrow">
      <span class="nm">${playerName(i)}</span>
      <span class="bar"><span style="width:${Math.round(c/max*100)}%"></span></span>
      <span class="ct">${c}</span>
    </div>`).join("");
}
